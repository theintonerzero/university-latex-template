\NewDocumentCommand{\newsubject}{m m m}{%
    \expandafter\def\csname subject#1.code\endcsname{#2}%
    \expandafter\def\csname subject#1.name\endcsname{#3}%
}

% Kept expandable so the values survive into the PDF metadata. Wrapping these
% in \NewDocumentCommand would make them \protected, and hyperref would write
% the raw key instead of the code and title.
\newcommand{\subjectcode}[1]{\csname subject#1.code\endcsname}
\newcommand{\subjectname}[1]{\csname subject#1.name\endcsname}
\newcommand{\subjectinfo}[1]{\subjectcode{#1}~--~\subjectname{#1}}

% Called once from main.tex. An unknown key would otherwise typeset nothing.
\NewDocumentCommand{\checksubject}{m}{%
    \ifcsname subject#1.code\endcsname\else
        \errmessage{Unknown subject key '#1'.
            Define it with \string\newsubject\space in utilities/utility.tex}%
    \fi
}
